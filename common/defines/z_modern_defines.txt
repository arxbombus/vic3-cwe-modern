### ================================================================
### Derived birthrate values
### ================================================================

### Birthrate at the end of the pre-transition segment
@hc_pg_normal_birthrate_at_transition = @[hc_pg_normal_max_birthrate*hc_pg_normal_transition_birthrate_mult]

### Birthrate slope from SoL 0 to transition SoL
@hc_pg_normal_birthrate_pretransition_slope = @[(hc_pg_normal_birthrate_at_transition-hc_pg_normal_max_birthrate)/hc_pg_normal_transition_sol]

### Rate where birthrate and mortality meet
@hc_pg_normal_rate_at_equilibrium = @[hc_pg_normal_equilibrium_sol*hc_pg_normal_birthrate_pretransition_slope+hc_pg_normal_max_birthrate]

### Birthrate at the mortality growth-max turning point
@hc_pg_normal_birthrate_at_growth_max = @[(hc_pg_normal_growth_max_sol-hc_pg_normal_transition_sol)*((hc_pg_normal_min_birthrate-hc_pg_normal_birthrate_at_transition)/(hc_pg_normal_stable_sol-hc_pg_normal_transition_sol))+hc_pg_normal_birthrate_at_transition]

### Birthrate slope from transition SoL to stable SoL
@hc_pg_normal_birthrate_transition_slope = @[(hc_pg_normal_min_birthrate-hc_pg_normal_birthrate_at_transition)/(hc_pg_normal_stable_sol-hc_pg_normal_transition_sol)]

### Birthrate intercept from transition SoL to stable SoL
@hc_pg_normal_birthrate_transition_intercept = @[-hc_pg_normal_birthrate_transition_slope*hc_pg_normal_stable_sol+hc_pg_normal_min_birthrate]


### ================================================================
### Derived mortality values
### ================================================================

### Mortality slope from SoL 0 to equilibrium SoL
@hc_pg_normal_mortality_starving_slope = @[(hc_pg_normal_rate_at_equilibrium-hc_pg_normal_max_mortality)/hc_pg_normal_equilibrium_sol]

### Mortality at the growth-max turning point
@hc_pg_normal_mortality_at_growth_max = @[hc_pg_normal_birthrate_at_growth_max*hc_pg_normal_max_growth_mortality_mult]

### Mortality slope from equilibrium SoL to growth-max SoL
@hc_pg_normal_mortality_equilibrium_to_growth_max_slope = @[(hc_pg_normal_mortality_at_growth_max-hc_pg_normal_rate_at_equilibrium)/(hc_pg_normal_growth_max_sol-hc_pg_normal_equilibrium_sol)]

### Mortality intercept from equilibrium SoL to growth-max SoL
@hc_pg_normal_mortality_equilibrium_to_growth_max_intercept = @[-hc_pg_normal_mortality_equilibrium_to_growth_max_slope*hc_pg_normal_equilibrium_sol+hc_pg_normal_rate_at_equilibrium]

### Mortality slope from growth-max SoL to stable SoL
@hc_pg_normal_mortality_growth_max_to_stable_slope = @[(hc_pg_normal_min_mortality-hc_pg_normal_mortality_at_growth_max)/(hc_pg_normal_stable_sol-hc_pg_normal_growth_max_sol)]

### Mortality intercept from growth-max SoL to stable SoL
@hc_pg_normal_mortality_growth_max_to_stable_intercept = @[-hc_pg_normal_mortality_growth_max_to_stable_slope*hc_pg_normal_stable_sol+hc_pg_normal_min_mortality]
