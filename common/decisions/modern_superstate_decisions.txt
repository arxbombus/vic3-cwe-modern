	is_shown = {
		is_player = yes
		c:USA ?= this
		NOT = { exists = c:ZGC }
	}

	when_taken = {
		save_scope_as = modern_superstate_founder
		every_country = {
			limit = {
				NOT = { THIS = ROOT }

					c:CAN ?= this

					c:QUE ?= this
					c:CSC ?= this

					c:WUS ?= this
					c:CUS ?= this
					c:SUS ?= this
					c:ASA ?= this
					c:TEX ?= this
				}
			}
			save_scope_as = modern_superstate_member
			scope:modern_superstate_founder = {
				annex_with_incorporation = scope:modern_superstate_member
			}
		}
		remove_modifier = modern_country_power_usa
		remove_modifier = modern_country_power_canada
		change_tag = ZGC
		add_primary_culture = cu:afro_american
		add_primary_culture = cu:anglo_canadian

		add_modifier = {

		}
